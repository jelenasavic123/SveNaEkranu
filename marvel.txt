    <div id="loading" class="loading" style="display: none;">Učitavanje...</div>

    <script>
        const apiKey = '2701858e9bccab54d4d0e4931420a739';
        const baseUrl = `https://api.themoviedb.org/3/`;
        const marvelKeywordId = 180547; // ID za Marvel
        const genreUrl = `${baseUrl}genre/movie/list?api_key=${apiKey}&language=sr-SR`;

        let currentPage = 1;
        let currentYear = '';
        let currentGenre = '';
        let isLoading = false;

        const yearFilters = document.getElementById('year-filters');
        const genreFilters = document.getElementById('genre-filters');

        // Generišemo dugmad za godine
        for (let year = 2024; year >= 2007; year--) {
            const button = document.createElement('button');
            button.classList.add('filter-button');
            button.textContent = year;
            button.onclick = () => selectYear(year);
            yearFilters.appendChild(button);
        }

        // Preuzimanje i generisanje dugmadi za žanrove
        fetch(genreUrl)
            .then(response => response.json())
            .then(data => {
                const genres = data.genres;
                genres.forEach(genre => {
                    const button = document.createElement('button');
                    button.classList.add('filter-button');
                    button.textContent = genre.name;
                    button.onclick = () => selectGenre(genre.id);
                    genreFilters.appendChild(button);
                });
            })
            .catch(error => console.error('Greška pri učitavanju žanrova:', error));

        function selectYear(year) {
            currentYear = year;
            currentPage = 1;
            fetchMovies();
        }

        function selectGenre(genreId) {
            currentGenre = genreId;
            currentPage = 1;
            fetchMovies();
        }

        function fetchMovies() {
            if (isLoading) return;

            isLoading = true;
            document.getElementById('movie-list').innerHTML = '';
            document.getElementById('loading').style.display = 'block';

            let url = `${baseUrl}discover/movie?api_key=${apiKey}&language=sr-SR&with_keywords=${marvelKeywordId}&page=${currentPage}`;

            if (currentYear) {
                url += `&primary_release_year=${currentYear}`;
            }

            if (currentGenre) {
                url += `&with_genres=${currentGenre}`;
            }

            fetch(url)
                .then(response => response.json())
                .then(data => {
                    const movies = data.results;
                    const movieList = document.getElementById('movie-list');

                    movies.forEach(movie => {
                        const movieItem = document.createElement('div');
                        movieItem.classList.add('movie-item');

                        const movieImage = document.createElement('img');
                        movieImage.src = `https://image.tmdb.org/t/p/w500${movie.poster_path}`;
                        movieImage.alt = movie.title;

                        const movieTitle = document.createElement('h3');
                        movieTitle.textContent = movie.title;

                        const movieRating = document.createElement('div');
                        movieRating.classList.add('rating');
                        movieRating.textContent = `TMDB: ${movie.vote_average}`;

                        movieItem.appendChild(movieImage);
                        movieItem.appendChild(movieTitle);
                        movieItem.appendChild(movieRating);
                        movieList.appendChild(movieItem);
                    });

                    currentPage++;
                    isLoading = false;
                    document.getElementById('loading').style.display = 'none';
                })
                .catch(error => {
                    console.error('Greška pri učitavanju filmova:', error);
                    isLoading = false;
                    document.getElementById('loading').style.display = 'none';
                });
        }
    </script>
