// Funkcija za povećanje jačine zvuka function increaseVolume() { const audioElement = document.querySelector('audio'); if (audioElement) { audioElement.volume = Math.min(audioElement.volume + 0.1, 1); console.log('Povećana zvučna jačina:', audioElement.volume); } } // Funkcija za smanjenje jačine zvuka function decreaseVolume() { const audioElement = document.querySelector('audio'); if (audioElement) { audioElement.volume = Math.max(audioElement.volume - 0.1, 0); console.log('Smanjena zvučna jačina:', audioElement.volume); } } // Funkcija za uključivanje/isključivanje titlova function toggleCaptions() { const videoElement = document.querySelector('video'); if (videoElement) { const tracks = videoElement.textTracks; for (let i = 0; i < tracks.length; i++) { tracks[i].mode = (tracks[i].mode === 'showing') ? 'none' : 'showing'; } console.log('Titlovi togglani:', tracks); } } // Funkcija za ulazak/izlazak iz fullscreen mod function toggleFullscreen() { const elem = document.documentElement; if (!document.fullscreenElement) { elem.requestFullscreen().catch(err => { console.error("Fullscreen error:", err); }); } else { document.exitFullscreen(); } } // Event listener za daljinski upravljač document.addEventListener('keydown', function(event) { switch (event.key) { case 'ArrowUp': increaseVolume(); break; case 'ArrowDown': decreaseVolume(); break; case 'c': toggleCaptions(); break; case 'f': toggleFullscreen(); break; default: console.log('Nepoznata komanda:', event.key); } }); // Add event listener for IR remote document.addEventListener('keydown', function(event) { if (event.key.match(/^KEY_/)) { // Regex to check IR keys switch (event.key) { case 'KEY_VOLUMEUP': increaseVolume(); break; case 'KEY_VOLUMEDOWN': decreaseVolume(); break; case 'KEY_CAPTIONS': toggleCaptions(); break; case 'KEY_FULLSCREEN': toggleFullscreen(); break; default: console.log('Nepoznata komanda IR:', event.key); } } });
