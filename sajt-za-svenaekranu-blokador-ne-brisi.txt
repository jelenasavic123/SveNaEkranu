    <!-- Početno je prazno, biće popunjeno kad korisnik klikne na dugme -->

<script>
    // Funkcija za prikazivanje sadržaja u iframe-u
    function prikaziSadrzaj(url) {
        var iframe = document.createElement('iframe');
        iframe.src = url;
        
        // Umetanje iframe-a u div
        document.getElementById('iframe-container').innerHTML = ''; // Resetujemo prethodni sadržaj
        document.getElementById('iframe-container').appendChild(iframe);
    }

    // Prikazivanje horoskopa odmah pri učitavanju stranice
    window.onload = function() {
        prikaziSadrzaj('https://horoskopapp.blogspot.com/');
    };

    // Dodavanje događaja za "Horoskop" dugme
    document.getElementById('horoskop-btn').addEventListener('click', function() {
        prikaziSadrzaj('https://horoskopapp.blogspot.com/');
    });

    // Dodavanje događaja za "TV Prognoza" dugme
    document.getElementById('tv-prognoza-btn').addEventListener('click', function() {
        prikaziSadrzaj('https://vremedanas.blogspot.com/');
    });

    // Dodavanje događaja za "Gatanje" dugme
    document.getElementById('gatanje-btn').addEventListener('click', function() {
        prikaziSadrzaj('https://gatanjesvenadlanu.blogspot.com/');
    });
</script>
